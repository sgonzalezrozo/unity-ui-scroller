#TMP_USING#using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// One item of the #NAME#ScrollList that opens when its hitbox is tapped. The hitbox asks the list to expand or collapse
/// the item; the list resizes it to Expanded Size, moves the items after it and sets IsExpanded, which shows the
/// details. The list remembers which items are open, so an item scrolled away and back comes back open.
/// Expanding works in lists, not in grids, bento or looping lists.
/// </summary>
public class #NAME#ItemController : MonoBehaviour, IExpandableSlot<#NAME#ItemModel>
{
    [SerializeField, Min(1f)] private float size = 120f;
    [SerializeField, Min(1f)] private float expandedSize = 320f;
    [SerializeField] private #TEXT_TYPE# title;
    [SerializeField] private #TEXT_TYPE# description;
    [SerializeField] private GameObject details;
    [SerializeField] private Button hitbox;

    private #NAME#ItemModel data;
    private bool isExpanded;

    public float DefaultExtent => size;

    public IScroller Owner { get; set; }
    public int ItemIndex { get; set; }

    public #NAME#ItemModel Data
    {
        get => data;
        set
        {
            data = value;
            title.text = value.Title;
            description.text = value.Description;
        }
    }

    public bool IsExpanded
    {
        get => isExpanded;
        set
        {
            isExpanded = value;
            details.SetActive(value);
        }
    }

    public void OnCollapsed()
        => IsExpanded = false;

    private void Awake()
        => hitbox.onClick.AddListener(Toggle);

    private void Toggle()
    {
        if (Owner == null)
            return;

        if (Owner.IsExpanded(ItemIndex))
            Owner.Collapse(ItemIndex);
        else
            Owner.Expand(ItemIndex, expandedSize);
    }
}
