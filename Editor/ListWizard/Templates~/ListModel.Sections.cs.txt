    /// <summary>
    /// Owns the #NAME# list's sections: a header and its items each. Change them with AddSection, RemoveSection and

    public class #NAME#ListModel : MonoBehaviour

        public SectionedSource<#NAME#ItemData, #NAME#HeaderData> Sections => sections;

        public void AddSection(#NAME#HeaderData header, IReadOnlyList<#NAME#ItemData> items)
        {
            if (header == null)
                throw new ArgumentNullException(nameof(header));
            if (items == null)
                throw new ArgumentNullException(nameof(items));

            sections.AddSection(header, new ListSource<#NAME#ItemData>(items));
            OnSectionsChanged?.Invoke();
        }

        public void RemoveSection(int sectionIndex)
        {
            sections.RemoveSection(sectionIndex);
            OnSectionsChanged?.Invoke();
        }

        public void ClearSections()
        {
            if (sections.SectionCount == 0)
                return;

            sections.ClearSections();
            OnSectionsChanged?.Invoke();

        {

                    items.Add(new #NAME#ItemData { #PLACEHOLDER_FIELDS# });

                #NAME#HeaderData header = new() { title = $"Section {section + 1}" };
                placeholders.AddSection(header, new ListSource<#NAME#ItemData>(items));
            }

        }
    }
