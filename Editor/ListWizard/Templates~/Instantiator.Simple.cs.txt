public class #NAME#Instantiator : MonoBehaviour
{
    [SerializeField] private #NAME#ScrollList scrollList;
    [Tooltip("Made-up items shown on Start until your code calls Show, so the list has something to show when you press Play. Set it to 0 once real data comes in.")]

    /// <summary>The items shown, in order.</summary>
    public IReadOnlyList<#NAME#ItemModel> Items => source.GetAllItems();

    private void Start()
    {
        if (!hasShown && placeholderCount > 0)
            Show(Placeholders(placeholderCount));
    }

    /// <summary>Shows these items, replacing the ones shown.</summary>
    public void Show(IEnumerable<#NAME#ItemModel> items)
    {
        source.SetItems(items);
        Apply();
    }

    public void Add(#NAME#ItemModel item)
    {
        source.AddItem(item);
        Apply();
    }

    public void Insert(int index, #NAME#ItemModel item)
    {
        source.InsertItem(index, item);
        Apply();
    }

    public void RemoveAt(int index)
    {
        source.RemoveItemAt(index);
        Apply();
    }

    /// <summary>Shows the current items again, after you changed fields of models already shown.</summary>
    public void Refresh()
        => Apply();

    private void Apply()
    {
        hasShown = true;
        if (scrollList.IsInitialized)
            scrollList.Refresh();
        else
            scrollList.Initialize(source);
    }
