#TMP_USING#using UnityEngine;
#UI_USING#
namespace #NAMESPACE#
{
    /// <summary>Renders one #NAME# list slot from its #NAME#ItemSlotModel: the summary always, the details while it's open.</summary>
    public class #NAME#ItemSlotView : MonoBehaviour
    {
        [SerializeField] private #TEXT_TYPE# titleText;
        [SerializeField] private #TEXT_TYPE# descriptionText;
        [SerializeField] private GameObject detailsRoot;

        private #NAME#ItemSlotModel slotModel;

        private void Awake()
        {
            slotModel = GetComponentInParent<#NAME#ItemSlotModel>();
        }

        private void OnEnable()
        {
            slotModel.OnItemDataChanged += RenderItem;
            RenderItem();
        }

        private void OnDisable()
        {
            slotModel.OnItemDataChanged -= RenderItem;
        }

        private void RenderItem()
        {
            detailsRoot.SetActive(slotModel.IsExpanded);

            #NAME#ItemData itemData = slotModel.Data;
            if (itemData == null)
                return;

            titleText.text = itemData.title;
            descriptionText.text = itemData.description;
        }
    }
}
