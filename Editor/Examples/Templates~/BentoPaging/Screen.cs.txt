using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// Shows the #NAME# bento from a PagedSource: the first page loads on start and the next ones as you scroll near the
/// end. The status label follows the loading. Drop the #NAME# prefab in a scene and press Play.
/// </summary>
public class #NAME#Screen : MonoBehaviour
{
    [SerializeField] private #NAME#Scroller scroller;
    [SerializeField] private Text status;

    private void Awake()
        => EnsureEventSystem();

    private void Start()
    {
        scroller.PageRequested += () => status.text = "Loading…";
        scroller.PageLoaded += _ => status.text = $"{scroller.GetItemCount()} tiles";
        scroller.PageFailed += exception => status.text = exception.Message;
        scroller.ReachedEnd += () => status.text = $"{scroller.GetItemCount()} tiles, no more pages";

        scroller.SetSource(new PagedSource<#NAME#Item>(new #NAME#PageLoader()));
    }
#EVENTSYSTEM#
}
