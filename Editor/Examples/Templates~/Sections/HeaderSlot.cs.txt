using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>A section header: the letter on the left and the number of contacts on the right.</summary>
public class #NAME#HeaderSlot : MonoBehaviour, ISlot<#NAME#Header>
{
    [SerializeField] private Text letter;
    [SerializeField] private Text count;

    private #NAME#Header data;

    public float DefaultExtent => 80f;

    public #NAME#Header Data
    {
        get => data;
        set
        {
            data = value;
            letter.text = value.Letter;
            count.text = value.Count == 1 ? "1 contact" : $"{value.Count} contacts";
        }
    }
}
