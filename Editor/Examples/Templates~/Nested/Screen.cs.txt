using System.Collections.Generic;
using UIScroller;
using UnityEngine;

/// <summary>
/// Builds made-up shelves of cards, like a store front. Drag sideways on a shelf to move it, up or down anywhere to
/// move the page. Drop the #NAME# prefab in a scene and press Play.
/// </summary>
public class #NAME#Screen : MonoBehaviour
{
    private static readonly string[] ShelfNames = { "Continue Playing", "Trending Now", "New Releases", "Because You Liked Puzzles", "Co-op Picks", "Hidden Gems" };
    private static readonly string[] Titles = { "Astro Drift", "Bloom", "Cinder", "Deep Echo", "Ember Run", "Fable Forge", "Glint", "Hollow Sky" };

    [SerializeField] private #NAME#Scroller scroller;
    [SerializeField] private int shelfCount = 30;
    [SerializeField] private int cardsPerShelf = 15;

    private void Awake()
        => EnsureEventSystem();

    private void Start()
    {
        List<#NAME#Shelf> shelves = new(shelfCount);
        for (int s = 0; s < shelfCount; s++)
        {
            List<#NAME#Item> cards = new(cardsPerShelf);
            for (int c = 0; c < cardsPerShelf; c++)
            {
                int id = s * cardsPerShelf + c;
                cards.Add(new #NAME#Item
                {
                    Title = Titles[id * 5 % Titles.Length],
                    Rating = 3f + id * 17 % 20 / 10f,
                    Color = Color.HSVToRGB(id * 0.618034f % 1f, 0.5f, 0.8f)
                });
            }

            shelves.Add(new #NAME#Shelf { Title = ShelfNames[s % ShelfNames.Length], Cards = new ListSource<#NAME#Item>(cards) });
        }

        scroller.Initialize(new ListSource<#NAME#Shelf>(shelves));
    }
#EVENTSYSTEM#
}
