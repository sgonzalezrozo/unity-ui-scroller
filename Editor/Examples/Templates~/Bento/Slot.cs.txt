using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// A bento tile. The list sizes it from the layout (1x1, 2x1, 2x2...), so everything inside is anchored, not sized:
/// the category at the top left, the value at the top right and the title at the bottom.
/// </summary>
public class #NAME#Slot : MonoBehaviour, ISlot<#NAME#Item>
{
    [SerializeField] private Image background;
    [SerializeField] private Text category;
    [SerializeField] private Text valueLabel;
    [SerializeField] private Text title;
    [SerializeField] private Button button;

    private #NAME#Item data;

    /// <summary>Ignored in bento: the cell size comes from the width, the columns and Cell Aspect.</summary>
    public float DefaultExtent => 240f;

    public #NAME#Item Data
    {
        get => data;
        set
        {
            data = value;
            background.color = value.Color;
            category.text = value.Category.ToUpperInvariant();
            valueLabel.text = value.Value.ToString();
            title.text = value.Title;
        }
    }

    private void Awake()
        => button.onClick.AddListener(() => Debug.Log($"[#NAME#] Open {data?.Title}"));
}
