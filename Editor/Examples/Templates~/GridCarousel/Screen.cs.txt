using System.Collections.Generic;
using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// Fills the #NAME# carousel and drives its page buttons and label. Swipe sideways to move a page at a time, or use
/// the arrows. Drop the #NAME# prefab in a scene and press Play.
/// </summary>
public class #NAME#Screen : MonoBehaviour
{
    private const float PAGE_ANIMATION_SECONDS = 0.35f;
    private static readonly string[] Names = { "Sword", "Shield", "Potion", "Bow", "Helmet", "Ring", "Scroll", "Boots", "Amulet", "Staff" };

    [SerializeField] private #NAME#Scroller scroller;
    [SerializeField] private Text pageLabel;
    [SerializeField] private Button previous;
    [SerializeField] private Button next;
    [SerializeField] private int itemCount = 60;

    private int page;

    private void Awake()
        => EnsureEventSystem();

    private void Start()
    {
        List<#NAME#Item> items = new(itemCount);
        for (int i = 0; i < itemCount; i++)
        {
            items.Add(new #NAME#Item
            {
                Name = $"{Names[i % Names.Length]} {i}",
                Price = 50 + i * 37 % 4950,
                Rarity = i * 13 % 5,
                Color = Color.HSVToRGB(i * 0.618034f % 1f, 0.5f, 0.75f)
            });
        }

        scroller.SnappedTo += OnSnapped;
        scroller.Rebuilt += UpdateLabel;
        previous.onClick.AddListener(() => GoToPage(page - 1));
        next.onClick.AddListener(() => GoToPage(page + 1));
        scroller.Initialize(new ListSource<#NAME#Item>(items));
    }

    /// <summary>Items on one page: the columns that fit in the viewport times the rows.</summary>
    private int ItemsPerPage()
    {
        float column = scroller.GetItemSize(0) + scroller.Settings.lineSpacing;
        int columns = column > 0f ? Mathf.Max(1, Mathf.FloorToInt((scroller.ViewportSize + scroller.Settings.lineSpacing) / column)) : 1;
        return columns * (int)Mathf.Max(1, scroller.Settings.slotsPerLine);
    }

    private int PageCount()
        => Mathf.Max(1, Mathf.CeilToInt(scroller.GetItemCount() / (float)ItemsPerPage()));

    private void GoToPage(int target)
    {
        page = Mathf.Clamp(target, 0, PageCount() - 1);
        scroller.ScrollToItem(page * ItemsPerPage(), ScrollAlign.Start, PAGE_ANIMATION_SECONDS);
        UpdateLabel();
    }

    private void OnSnapped(int index)
    {
        page = index / ItemsPerPage();
        UpdateLabel();
    }

    private void UpdateLabel()
    {
        pageLabel.text = $"Page {page + 1} / {PageCount()}";
        previous.interactable = page > 0;
        next.interactable = page < PageCount() - 1;
    }
#EVENTSYSTEM#
}
