using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>A tile in a horizontal grid page: artwork, a rarity strip, a name and a price.</summary>
public class #NAME#Slot : MonoBehaviour, ISlot<#NAME#Item>
{
    private static readonly Color[] RarityColors =
    {
        new(0.62f, 0.65f, 0.7f), new(0.35f, 0.75f, 0.4f), new(0.3f, 0.55f, 0.95f), new(0.7f, 0.4f, 0.95f), new(0.98f, 0.7f, 0.2f)
    };

    [SerializeField] private Image art;
    [SerializeField] private Text icon;
    [SerializeField] private Image rarityStrip;
    [SerializeField] private Text itemName;
    [SerializeField] private Text price;
    [SerializeField] private Button button;

    private #NAME#Item data;

    /// <summary>The width of a column (the scroll axis is horizontal). Its height comes from the number of rows.</summary>
    public float DefaultExtent => 300f;

    public #NAME#Item Data
    {
        get => data;
        set
        {
            data = value;
            art.color = value.Color;
            icon.text = value.Name.Substring(0, 1);
            rarityStrip.color = RarityColors[Mathf.Clamp(value.Rarity, 0, RarityColors.Length - 1)];
            itemName.text = value.Name;
            price.text = $"{value.Price:N0} G";
        }
    }

    private void Awake()
        => button.onClick.AddListener(() => Debug.Log($"[#NAME#] Buy {data?.Name}"));
}
