using System.Collections.Generic;
using UIScroller;
using UnityEngine;

/// <summary>Fills the #NAME# list with heroes. Tap a row's summary to open it. Drop the #NAME# prefab in a scene and press Play.</summary>
public class #NAME#Screen : MonoBehaviour
{
    private static readonly string[] Names = { "Aria", "Bram", "Cleo", "Dax", "Elio", "Fenn", "Gale", "Hux" };
    private static readonly string[] Roles = { "Tank", "Healer", "Ranger", "Mage", "Rogue" };

    [SerializeField] private #NAME#Scroller scroller;
    [SerializeField] private int itemCount = 200;

    private void Awake()
        => EnsureEventSystem();

    private void Start()
    {
        List<#NAME#Item> items = new(itemCount);
        for (int i = 0; i < itemCount; i++)
        {
            items.Add(new #NAME#Item
            {
                Name = $"{Names[i % Names.Length]} {i}",
                Role = Roles[i * 3 % Roles.Length],
                Level = 1 + i * 7 % 60,
                Attack = i * 37 % 100 / 100f,
                Defense = i * 53 % 100 / 100f,
                Speed = i * 71 % 100 / 100f,
                Color = Color.HSVToRGB(i * 0.618034f % 1f, 0.5f, 0.8f)
            });
        }

        scroller.Initialize(new ListSource<#NAME#Item>(items));
    }
#EVENTSYSTEM#
}
