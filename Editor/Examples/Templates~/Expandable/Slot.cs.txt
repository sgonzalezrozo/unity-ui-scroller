using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// A row with a fixed height that opens to EXPANDED_HEIGHT. The Hitbox button covers only the summary at the top: it
/// asks the list (the model of what's open) to expand or collapse this row. The list resizes the row, moves the rows
/// below and sets IsExpanded, which shows the details. Buttons inside the details keep working on their own.
/// </summary>
public class #NAME#Slot : MonoBehaviour, IExpandableSlot<#NAME#Item>
{
    public const float HEIGHT = 170f;
    public const float EXPANDED_HEIGHT = 520f;

    [SerializeField] private Image portrait;
    [SerializeField] private Text initial;
    [SerializeField] private Text heroName;
    [SerializeField] private Text role;
    [SerializeField] private Text level;
    [SerializeField] private Text chevron;
    [SerializeField] private Button hitbox;
    [SerializeField] private GameObject details;
    [SerializeField] private RectTransform attackFill;
    [SerializeField] private RectTransform defenseFill;
    [SerializeField] private RectTransform speedFill;
    [SerializeField] private Button selectButton;

    private #NAME#Item data;
    private bool isExpanded;

    public float DefaultExtent => HEIGHT;

    public IScroller Owner { get; set; }
    public int ItemIndex { get; set; }

    public #NAME#Item Data
    {
        get => data;
        set
        {
            data = value;
            portrait.color = value.Color;
            initial.text = value.Name.Substring(0, 1);
            heroName.text = value.Name;
            role.text = value.Role;
            level.text = $"Lv {value.Level}";
            SetFill(attackFill, value.Attack);
            SetFill(defenseFill, value.Defense);
            SetFill(speedFill, value.Speed);
        }
    }

    public bool IsExpanded
    {
        get => isExpanded;
        set
        {
            isExpanded = value;
            details.SetActive(value);
            chevron.text = value ? "▲" : "▼";
        }
    }

    public void OnCollapsed()
        => IsExpanded = false;

    private void Awake()
    {
        hitbox.onClick.AddListener(Toggle);
        selectButton.onClick.AddListener(() => Debug.Log($"[#NAME#] Selected {data?.Name}"));
    }

    private void Toggle()
    {
        if (Owner == null)
            return;

        if (Owner.IsExpanded(ItemIndex))
            Owner.Collapse(ItemIndex);
        else
            Owner.Expand(ItemIndex, EXPANDED_HEIGHT);
    }

    /// <summary>A stat bar: the fill's right anchor at <paramref name="value"/> (0 to 1) of the bar.</summary>
    private static void SetFill(RectTransform fill, float value)
        => fill.anchorMax = new Vector2(Mathf.Clamp01(value), 1f);
}
