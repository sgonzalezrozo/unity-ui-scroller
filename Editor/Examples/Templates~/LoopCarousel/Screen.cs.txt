using System.Collections.Generic;
using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// Fills the #NAME# carousel and shows which card is centred. Drag sideways: it loops both ways and snaps a card to
/// the middle. Drop the #NAME# prefab in a scene and press Play.
/// </summary>
public class #NAME#Screen : MonoBehaviour
{
    private static readonly string[] Worlds = { "Ember Peaks", "Frost Hollow", "Verdant Reach", "Sunken City", "Sky Docks", "Ash Wastes", "Moon Garden" };

    [SerializeField] private #NAME#Scroller scroller;
    [SerializeField] private Text caption;
    [SerializeField] private int itemCount = 14;

    private List<#NAME#Item> items;

    private void Awake()
        => EnsureEventSystem();

    private void Start()
    {
        items = new List<#NAME#Item>(itemCount);
        for (int i = 0; i < itemCount; i++)
        {
            items.Add(new #NAME#Item
            {
                Title = Worlds[i % Worlds.Length],
                Subtitle = $"World {i + 1} · {3 + i * 5 % 12} levels",
                Color = Color.HSVToRGB(i * 0.618034f % 1f, 0.55f, 0.8f)
            });
        }

        scroller.SnappedTo += ShowCaption;
        scroller.Initialize(new ListSource<#NAME#Item>(items));
        ShowCaption(0);
    }

    private void ShowCaption(int index)
    {
        int wrapped = (index % items.Count + items.Count) % items.Count;
        caption.text = $"{items[wrapped].Title}  ({wrapped + 1}/{items.Count})";
    }
#EVENTSYSTEM#
}
