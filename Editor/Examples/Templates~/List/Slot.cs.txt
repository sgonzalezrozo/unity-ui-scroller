using UIScroller;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// A row of the #NAME# list: an avatar with the item's initial, a title, a subtitle and an unread badge. The list
/// reuses rows as you scroll, so the Data setter sets everything the row shows.
/// </summary>
public class #NAME#Slot : MonoBehaviour, ISlot<#NAME#Item>
{
    [SerializeField] private Image avatar;
    [SerializeField] private Text initial;
    [SerializeField] private Text title;
    [SerializeField] private Text subtitle;
    [SerializeField] private GameObject badge;
    [SerializeField] private Text badgeLabel;
    [SerializeField] private Button button;

    private #NAME#Item data;

    /// <summary>Every row's height, in pixels.</summary>
    public float DefaultExtent => 150f;

    public #NAME#Item Data
    {
        get => data;
        set
        {
            data = value;
            avatar.color = value.Color;
            initial.text = string.IsNullOrEmpty(value.Title) ? "?" : value.Title.Substring(0, 1);
            title.text = value.Title;
            subtitle.text = value.Subtitle;
            badge.SetActive(value.Unread > 0);
            badgeLabel.text = value.Unread > 99 ? "99+" : value.Unread.ToString();
        }
    }

    private void Awake()
        => button.onClick.AddListener(OnClick);

    private void OnClick()
        => Debug.Log($"[#NAME#] Clicked {data?.Title}");
}
